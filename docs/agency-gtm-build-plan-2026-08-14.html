import { NextResponse } from "next/server";

import prisma from "@/lib/prisma";
import { openCronRun, closeCronRun } from "@/app/api/cron/_lib/run";
import { sendLifecycleStep } from "@/modules/agency-portal/notify/lifecycle";

// Day 2 · day 5 · day 10 after the agency's FIRST enrichment run.
const STEPS = [
  { step: 1, afterDays: 2 },
  { step: 2, afterDays: 5 },
  { step: 3, afterDays: 10 },
] as const;

export async function GET(req: Request) {
  if (req.headers.get("authorization") !== `Bearer ${process.env.CRON_SECRET}`) {
    return new Response("Unauthorized", { status: 401 });
  }

  const run = await openCronRun("daily:lifecycle-nudge");
  let sent = 0;

  try {
    for (const { step, afterDays } of STEPS) {
      const cutoff = new Date(Date.now() - afterDays * 86_400_000);

      // Agencies whose first run is older than the cutoff, that have NOT
      // come back since, and that have not already had this step.
      const candidates = await prisma.agency.findMany({
        where: {
          createdAt: { lt: cutoff },
          events: { none: { type: "lifecycle_sent", propsJson: { path: ["step"], equals: step } } },
          runs: { some: {} },
        },
        select: { id: true, slug: true, members: { select: { user: { select: { email: true } } }, take: 1 } },
        take: 100, // bounded per run — .claude/rules/scalability.md
      });

      for (const agency of candidates) {
        const email = agency.members[0]?.user.email;
        if (!email) continue;
        await sendLifecycleStep({ agencyId: agency.id, email, step });
        await prisma.productEvent.create({
          data: { agencyId: agency.id, type: "lifecycle_sent", propsJson: { step } },
        });
        sent++;
      }
    }

    await closeCronRun(run.id, "OK", sent);
    return NextResponse.json({ ok: true, sent });
  } catch (e) {
    await closeCronRun(run.id, "FAILED", sent, String(e));
    throw e;
  }
}

/**
 * Activation · an agency that ran enrichment on 2+ distinct calendar days.
 * Chosen because the single retained agency (oleksii-koshkarov) crossed it
 * and all five churned agencies did not.
 */
export async function getActivationRate(): Promise<{
  signed: number;
  activated: number;
  rate: number;
}> {
  const rows = await prisma.$queryRaw<{ agencyId: string; days: bigint }[]>`
    SELECT "agencyId", COUNT(DISTINCT DATE("startedAt")) AS days
    FROM "EnrichmentRun"
    GROUP BY "agencyId"
  `;
  const signed = rows.length;
  const activated = rows.filter((r) => Number(r.days) >= 2).length;
  return { signed, activated, rate: signed ? activated / signed : 0 };
}

/**
 * Markets with enough enriched depth to make a good first impression.
 * Threshold: >= 200 businesses AND >= 25% carrying a BusinessSnapshot.
 * Recomputed weekly; cached for a day.
 */
export async function getSeededMarkets(limit = 6) {
  "use cache";
  cacheLife("days");
  cacheTag("seeded-markets");
  if (process.env.NEXT_PHASE === "phase-production-build") return EMPTY_SEEDED;

  try {
    return await prisma.$queryRaw`
      SELECT b.country, b.category, b.city, COUNT(*)::int AS n
      FROM "Business" b
      WHERE b."isActive"
      GROUP BY 1, 2, 3
      HAVING COUNT(*) >= 200
      ORDER BY n DESC
      LIMIT ${limit}
    `;
  } catch {
    return EMPTY_SEEDED;
  }
}
